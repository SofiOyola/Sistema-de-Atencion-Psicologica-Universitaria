// 2. CREACION DE DATOS (10–25 REGISTROS POR ENTIDAD)

// =====================================================
// 3. RELACIONES (MEZCLADAS Y COHERENTES)
// =====================================================
// 3. RELACIONES 
// 1. SOLICITA  Estudiante — Cita
MATCH (e:Estudiante), (c:Cita)
WHERE e.id_estudiante = c.id_cita
   OR (c.id_cita % 5) = (e.id_estudiante % 5)
CREATE (e)-[:SOLICITA]->(c);

// 2. ATIENDE  Psicologo — Cita
MATCH (p:Psicologo), (c:Cita)
WHERE (c.id_cita % 10) + 1 = p.id_psicologo
CREATE (p)-[:ATIENDE]->(c);

// 3. POSEE  Estudiante — Historial_Clinico
MATCH (e:Estudiante), (h:Historial_Clinico)
WHERE e.id_estudiante = h.id_historial
CREATE (e)-[:POSEE]->(h);

// 4. CONTIENE  Historial_Clinico — Nota_Seguimiento
MATCH (h:Historial_Clinico), (n:Nota_Seguimiento)
WHERE (n.id_nota % 15) + 1 = h.id_historial
CREATE (h)-[:CONTIENE]->(n);

// 5. REGISTRA  Psicologo — Nota_Seguimiento
MATCH (p:Psicologo), (n:Nota_Seguimiento)
WHERE (n.id_nota % 10) + 1 = p.id_psicologo
CREATE (p)-[:REGISTRA]->(n);

// 6. GENERA  Cita — Nota_Seguimiento
MATCH (c:Cita), (n:Nota_Seguimiento)
WHERE (n.id_nota % 20) + 1 = c.id_cita
CREATE (c)-[:GENERA]->(n);

// 7. MODIFICA  Cita — Modificacion_Cita
MATCH (c:Cita), (m:Modificacion_Cita)
WHERE (m.id_modificacion % 10) + 1 = c.id_cita
CREATE (c)-[:MODIFICA]->(m);

// 8. TIENE_DISPONIBILIDAD  Psicologo — Disponibilidad
MATCH (p:Psicologo), (d:Disponibilidad)
WHERE (p.id_psicologo % 8) + 1 = d.id_disponibilidad
CREATE (p)-[:TIENE_DISPONIBILIDAD]->(d);

// 9. REPORTA  Estudiante — Estado_Emocional
MATCH (e:Estudiante), (s:Estado_Emocional)
WHERE (s.id_estado_emocional % 15) + 1 = e.id_estudiante
CREATE (e)-[:REPORTA]->(s);

// 10. GENERA_ALERTA  Estado_Emocional — Alerta_Emocional
MATCH (s:Estado_Emocional), (a:Alerta_Emocional)
WHERE (a.id_alerta % 25) + 1 = s.id_estado_emocional
CREATE (s)-[:GENERA_ALERTA]->(a);

// 11. CREA  Psicologo — Recurso_Psicoeducativo
MATCH (p:Psicologo), (r:Recurso_Psicoeducativo)
WHERE (r.id_recurso % 10) + 1 = p.id_psicologo
CREATE (p)-[:CREA]->(r);

// 12. TIENE  Administrador — Usuario, Estudiante – Usuario, Psicologo – Usuario

// Admin → usuarios 1–5
MATCH (ad:Administrador), (u:Usuario)
WHERE u.id_usuario <= 5
  AND ad.id_administrativo = u.id_usuario
CREATE (ad)-[:TIENE]->(u);

// Estudiante → usuarios 6–15
MATCH (e:Estudiante), (u:Usuario)
WHERE u.id_usuario >= 6 AND u.id_usuario <= 15
  AND (u.id_usuario - 5) = e.id_estudiante
CREATE (e)-[:TIENE]->(u);

// Psicologo → usuarios 16–20
MATCH (p:Psicologo), (u:Usuario)
WHERE u.id_usuario >= 16 AND u.id_usuario <= 20
  AND (u.id_usuario - 15) = p.id_psicologo
CREATE (p)-[:TIENE]->(u);

// 13. REALIZA  Usuario — Trazabilidad
MATCH (u:Usuario), (t:Trazabilidad)
WHERE (t.id_trazabilidad % 20) + 1 = u.id_usuario
CREATE (u)-[:REALIZA]->(t);

// 14. ASIGNA  Estudiante — Asignacion
MATCH (e:Estudiante), (a:Asignacion)
WHERE (a.id_asignacion % 15) + 1 = e.id_estudiante
CREATE (e)-[:ASIGNA]->(a);

// 15. CORRESPONDE  Psicologo — Asignacion
MATCH (p:Psicologo), (a:Asignacion)
WHERE (a.id_asignacion % 10) + 1 = p.id_psicologo
CREATE (p)-[:CORRESPONDE]->(a);

// =====================================================
// 4. VISUALIZAR
// =====================================================
MATCH (n) RETURN n;
