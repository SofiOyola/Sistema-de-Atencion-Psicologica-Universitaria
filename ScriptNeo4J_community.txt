// 3. RELACIONES

MATCH (e:Estudiante), (c:Cita)
WHERE e.id_estudiante = c.id_cita
   OR (c.id_cita % 5) = (e.id_estudiante % 5)
CREATE (e)-[:SOLICITA]->(c);

MATCH (p:Psicologo), (c:Cita)
WHERE ((c.id_cita - 1) % 10) + 1 = p.id_psicologo
CREATE (p)-[:ATIENDE]->(c);

MATCH (e:Estudiante), (h:Historial_Clinico)
WHERE e.id_estudiante = h.id_historial
CREATE (e)-[:POSEE]->(h);

MATCH (h:Historial_Clinico), (n:Nota_Seguimiento)
WHERE ((n.id_nota - 1) % 15) + 1 = h.id_historial
CREATE (h)-[:CONTIENE]->(n);

MATCH (p:Psicologo), (n:Nota_Seguimiento)
WHERE ((n.id_nota - 1) % 10) + 1 = p.id_psicologo
CREATE (p)-[:REGISTRA]->(n);

MATCH (c:Cita), (n:Nota_Seguimiento)
WHERE ((n.id_nota - 1) % 20) + 1 = c.id_cita
CREATE (c)-[:GENERA]->(n);

MATCH (c:Cita), (m:Modificacion_Cita)
WHERE ((m.id_modificacion - 1) % 10) + 1 = c.id_cita
CREATE (c)-[:MODIFICA]->(m);

MATCH (p:Psicologo), (d:Disponibilidad)
WHERE ((p.id_psicologo - 1) % 8) + 1 = d.id_disponibilidad
CREATE (p)-[:TIENE_DISPONIBILIDAD]->(d);

MATCH (e:Estudiante), (s:Estado_Emocional)
WHERE ((s.id_estado_emocional - 1) % 15) + 1 = e.id_estudiante
CREATE (e)-[:REPORTA]->(s);

MATCH (s:Estado_Emocional), (a:Alerta_Emocional)
WHERE ((a.id_alerta - 1) % 25) + 1 = s.id_estado_emocional
CREATE (s)-[:GENERA_ALERTA]->(a);

MATCH (p:Psicologo), (r:Recurso_Psicoeducativo)
WHERE ((r.id_recurso - 1) % 10) + 1 = p.id_psicologo
CREATE (p)-[:CREA]->(r);

MATCH (u:Usuario), (t:Trazabilidad)
WHERE ((t.id_trazabilidad - 1) % 30) + 1 = u.id_usuario
CREATE (u)-[:REALIZA]->(t);

MATCH (e:Estudiante), (a:Asignacion)
WHERE ((a.id_asignacion - 1) % 15) + 1 = e.id_estudiante
CREATE (e)-[:ASIGNA]->(a);

MATCH (p:Psicologo), (a:Asignacion)
WHERE ((a.id_asignacion - 1) % 10) + 1 = p.id_psicologo
CREATE (p)-[:CORRESPONDE]->(a);

// 4. VERIFICACION RAPIDA
MATCH (u:User)
OPTIONAL MATCH (e:Estudiante)-[:TIENE]->(u)
OPTIONAL MATCH (p:Psicologo)-[:TIENE]->(u)
OPTIONAL MATCH (a:Administrador)-[:TIENE]->(u)
RETURN u.email AS email,
       u.role AS role,
       e.id_estudiante AS studentId,
       p.id_psicologo AS psychologistId,
       a.id_administrativo AS adminId
ORDER BY u.id_usuario;
